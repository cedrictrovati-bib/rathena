pay_fild04,34,177,5	script	Terrified Poring#kpg_ent	1002,{
	set .@party_id, getcharid(1);
	set .@md_name$, "King Poring Gluttony";

	mes "[Terrified Poring]";
	mes "Pli... Pli... help us!";
	next;

	if (instance_id() == 0) {
		if (checkquest(40401, PLAYTIME) == 0 || checkquest(40401, PLAYTIME) == 1) {
			mes "The King is resting. Come back later.";
			close;
		}
	}

	if (instance_id()) set .@text$, "Rejoin Instance";
	else set .@text$, "Create Instance";

	switch(select("Stop the Gluttony ("+.@text$+"):Leave")) {
		case 1:
			if (instance_id() == 0) {
				if (.@party_id <= 0) { mes "You must be in a party."; close; }
				if (getcharid(0) != getpartyleader(.@party_id, 2)) { mes "Only the leader can start."; close; }
				if (instance_create(.@md_name$) < 0) { mes "Failed to create."; close; }
				if (checkquest(40401) == -1) setquest 40401;
			}
			instance_enter(.@md_name$);
			end;
		case 2: close;
	}
}
