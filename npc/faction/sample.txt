prontera,160,180,3	script	Faction Sample	100,{
mes "[Faction Sample]", "Hello!", "I'm the sample NPC for faction system script commands";
next;
mes "[Faction Sample]", "Let's start...", "Input faction ID 1~3 and i'll make you a faction member";
next;
input(@faction_id,1,3);
mes "[Faction Sample]","You choose '"+factioninfo(@faction_id,0)+"' faction:","What are you want?";
next;
L_Menu:
mes "[Faction Sample]";
	switch(select("Give me more Info about this faction:Make me a faction member of '"+factioninfo(@faction_id,0)+"':FvF On:FvF Off:Spawn Faction Monster:Set faction leader:Vote Start:Vote Stop:Vote to someone:Relic add:Relic info:Relic enable:Relic disable:Relic delete:Teleport to faction choice"))
	{
		case 1: callsub(S_FACTION_INFO); break;
		case 2: callsub(S_FACTION_SET); break;
		case 3: callsub(S_FACTION_FVF_SET,1); break;
		case 4: callsub(S_FACTION_FVF_SET,0); break;
		case 5: callsub(S_FACTION_SPAWN); break;
		case 6: callsub(S_FACTION_LEADER); break;
		case 7: callsub(S_VOTE_START); break;
		case 8: callsub(S_VOTE_STOP); break;
		case 9: callsub(S_VOTE_SOMEONE); break;
		case 10: callsub(S_RELIC_ADD); break;
		case 11: callsub(S_RELIC_INFO); break;
		case 12: callsub(S_RELIC_ENABLE); break;
		case 13: callsub(S_RELIC_DISABLE); break;
		case 14: callsub(S_RELIC_DEL); break;		
		case 15: callsub(S_FACTION_TP); break;
	}
close;

S_FACTION_SPAWN:
	factionmonster(@faction_id,"this",159,180,factioninfo(@faction_id,1),1002,1);
	close;

S_FACTION_FVF_SET:
	.@map$ = strnpcinfo(4);
	.@s = getmapflag(.@map$,mf_fvf);
	switch(getarg(0))
	{
		case 1: if( !.@s ) fvfon(.@map$); break;
		default: if( .@s ) fvfoff(.@map$); break;
	}
	close;

S_FACTION_SET:
	setfaction(@faction_id);
	mes "Now you're in Faction '"+factioninfo(@faction_id,0)+"'";
	close;
	
S_FACTION_TP:
	warp "job3_rune02",38,26;
	close;

S_FACTION_LEADER:
	setfactionleader(@faction_id,getcharid(0));
	mes "Now you're the Faction Leader '"+factioninfo(@faction_id,0)+"'";
	close;

S_VOTE_START:
	votingstart(@faction_id);
	close;

S_VOTE_STOP:
	votingstop(@faction_id);
	close;

S_VOTE_SOMEONE:
	vote(getcharid(0),1); //for yourself
	close;

S_RELIC_ADD:
	relicadd(@faction_id,1161,1);
	close;

S_RELIC_INFO:
	input(@relic_slot,0,4);
	mes "Relic info : " + relicgetinfo(@faction_id,@relic_slot);
	close;

S_RELIC_ENABLE: //need fvfon with args like @fvfon <faction id> <relic id>
	input(@relic_slot,0,4);
	relicactivate(@faction_id,@relic_slot,1);
	close;

S_RELIC_DISABLE:
	input(@relic_slot,0,4);
	relicactivate(@faction_id,@relic_slot,0);
	close;

S_RELIC_DEL:
	input(@relic_slot,0,4);
	relicdel(@faction_id,@relic_slot);
	close;

S_FACTION_INFO:
	mes "Faction id:" +@faction_id;
	mes "Name:" +factioninfo(@faction_id,0);
	mes "Unit name:" +factioninfo(@faction_id,1);
	mes "Home map:" +factioninfo(@faction_id,2);
	mes "Home map X:" +factioninfo(@faction_id,3);
	mes "Home map Y:" +factioninfo(@faction_id,4);
	mes "Race:" +factioninfo(@faction_id,5);
	mes "Element:" +factioninfo(@faction_id,6);
	mes "Element lvl:" +factioninfo(@faction_id,7);
	mes "Size:" +factioninfo(@faction_id,8);
	mes "Clothes color:" +factioninfo(@faction_id,9);
	mes "Hair color:" +factioninfo(@faction_id,10);
	mes "Hair style:" +factioninfo(@faction_id,11);
	mes "Voting state:" +factioninfo(@faction_id,12);
	mes "Alliances :";
	explode(.@alliances$,factioninfo(@faction_id,13),":");
	for(.@i=0;.@i<getarraysize(.@alliances$);.@i++)
		mes (.@i + 1) + ") " + .@alliances$[.@i];
	mes "Auras :";
	explode(.@auras$,factioninfo(@faction_id,14),":");
	for(.@i=0;.@i<getarraysize(.@auras$);.@i++)
		mes (.@i + 1) + ") " + .@auras$[.@i];
	mes "RP race:" +factioninfo(@faction_id,15);

	next;
	goto L_Menu;
}

job3_rune02,38,42,3	script	Prism of Fate	10547,{
	// Define the Stone's name as a constant
    .stone_name$ = "Prism of Fate";
	
	mes "[ ^FF0000" + .stone_name$ + "^000000 ]";
	mes "A strange, humming vibration fills the air. As you approach the crystal, your reflection seems to split into two distinct forms.";
	next;
		
	if (Class == Job_Novice) {
		mes "[ ^FF0000" + .stone_name$ + "^000000 ]";
		mes "The Prism remains dark. It does not recognize your potential yet.";
		mes "Return once you have found your path as an adventurer.";
		close;
	}

	if (@faction_id > 1) {
		mes "[ ^FF0000" + .stone_name$ + "^000000 ]";
		mes "The Prism has already read your soul. Your path is set.";
		close;
	}
	
	mes "[ ^FF0000" + .stone_name$ + "^000000 ]";
	mes "Which side of the spectrum do you belong to?";
	next;

	switch(select("The Aetherials - The Celestial Light.:The Remnant - The Earthly Shadows.:Do not touch the stone.")) {
		    
		case 1: // Aetherials
		    mes "[ ^0000FFThe Aetherials^000000 ]";
		    mes "You feel a surge of warmth. You see visions of Asgard and the pure light of the Valkyries.";
		    mes "Do you swear to purge the corruption and uphold the Divine Order?";
		    next;
		    if(select("I swear.:I am not ready.") == 2) goto OnCancel;
		        
		    set @faction_id, 2;
            setfaction(@faction_id);
		    specialeffect2 EF_HEAL2; // Golden Holy effect
		    mes "The Prism glows with a blinding gold light.";
		    mes "Welcome, Aetherial.";
		    announce strcharinfo(0) + " has been sanctified as an Aetherial!", bc_all;
			warp "hero_ent2",90,79; // Warp to Aetherial area
			getitem 23360, 1;
			dispbottom "You received Aetherials city scroll.";

		    close;

		case 2: // Remnant
		    mes "[ ^660099The Remnant^000000 ]";
		    mes "You feel a cold, haunting resonance. You see the flickering shadows of the Bio-Labs and the scars of the fallen.";
		    mes "Do you embrace your survival and the truth of the shadows?";
		    next;
		    if(select("I embrace it.:I am not ready.") == 2) goto OnCancel;
				
		    set @faction_id, 3;
            setfaction(@faction_id);
		    specialeffect2 EF_PHANTOM_BARD; // Purple/Ghostly effect
		    mes "The Prism pulses with a deep, violet energy.";
		    mes "Welcome, Remnant.";
		    announce strcharinfo(0) + " has emerged as a Remnant!", bc_all;
			warp "nif_dun01",146,161; // Warp to Remnant area
			getitem 23361, 1;
			dispbottom "You received Remnants city scroll.";
		    close;

		case 3:
		OnCancel:
		    mes "[ ^FF0000" + .stone_name$ + "^000000 ]";
		    mes "The visions fade. The Prism waits for your destiny.";
		    close;

	OnInit:
		setnpcdisplay("Prism of Fate", "Prism of Fate", 10547, 2);
		end;
	}
}
